\begin{algorithm}[H]
\DontPrintSemicolon
\caption{LLM-Guided Object Search}
\label{alg:main}
\Input{target category $c^{*}$; RGB-D observations $(I_t, D_t)$ and pose $x_t$; step budget $T$}
\Output{\Stop at the target (\Found), or \Fail when the budget is spent}
\State{scene graph $\mathcal{G}$; candidate nodes $\mathcal{N}$; distances $d$; probabilities $p$; visit order $\pi$; destination $v$; path $\tau$}
\BlankLine
\Comment{Warm-up}
\WarmUp{} \Comment*[r]{rotate $360^{\circ}$ in place: first view of the surroundings}
$\mathcal{G} \gets \InitSceneGraph{$x_0$}$ \Comment*[r]{initial map, rooms and objects from the warm-up scan}
\While{$t < T$}{
  \Comment{Scene graph}
  $\mathcal{G} \gets \UpdateSceneGraph{$\mathcal{G}, I_t, D_t, x_t$}$ \Comment*[r]{map, rooms, openings, detected objects}
  $\mathcal{G} \gets \ClassifyRooms{$\mathcal{G}$}$ \Comment*[r]{LLM assigns each room a type from the objects seen in it}
  $\mathcal{N} \gets \CandidateNodes{$\mathcal{G}$}$ \Comment*[r]{unscanned rooms, openings, sighted targets}
  \lIf(\Comment*[f]{nothing to plan over}){$\mathcal{N} = \varnothing$}{\ExploreFrontier{}; \KwContinue}
  
  \Comment{Distances}
  $d \gets \Dijkstra{$\mathcal{G}, x_t, \mathcal{N}$}$ \Comment*[r]{travel cost from $x_t$ to, and between, the nodes}
  
  \Comment{Probabilities}
  $p \gets \LLM{$c^{*}, \mathcal{G}, \mathcal{N}$}$ \Comment*[r]{$p_i$: probability that the target is found at node $i$}
  
  \Comment{Visit order}
  $\pi \gets \RPT{$\mathcal{N}, p, d$}$;\quad $v \gets \pi_1$ \Comment*[r]{order minimising the expected time-to-find}
  
  \Comment{Navigation, local exploration and verification}
  \Repeat{$v$ explored \KwOr $c^{*}$ is sighted}{
    \uIf{not yet arrived at $v$}{
      $\tau \gets \AStar{$x_t, v$}$ \Comment*[r]{collision-free path to $v$ on the current map}
      \Navigate{$\tau$} \Comment*[r]{one step along $\tau$; perception keeps updating $\mathcal{G}$}
    }
    \Else{
      \ExploreLocal{$v$} \Comment*[r]{room: walk to its vantage point and scan $360^{\circ}$; opening: peek from the threshold}
    }
  }
  \uIf{$c^{*}$ is sighted}{
    \lIf(\Comment*[f]{verify, approach by A$^{*}$, inspect; else release and re-plan}){\TargetClosing{$c^{*}$} $=$ \Stop}{\KwRet{\Found}}
  }
  \lElse(\Comment*[f]{the node is done; the next iteration re-plans}){mark $v$ explored}
}
\KwRet{\Fail}
\end{algorithm}
