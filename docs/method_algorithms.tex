\section*{Method pseudocode and notation}

The method is a per-action loop (Algorithm~\ref{alg:main}) over eight designed
sub-components: an observed-only scene graph (Alg.~\ref{alg:scenegraph}), a
position-based scan ledger that decides when a room is finished
(Alg.~\ref{alg:finished}), the candidate-node set of rooms, openings, target
landmarks and staircases (Alg.~\ref{alg:nodes}), an LLM oracle that estimates
per-node find probabilities (Alg.~\ref{alg:oracle}), a cost-aware visit order
solved by RPT$^{\ast}$ (Alg.~\ref{alg:order}), the one-scan / one-peek visit
primitives (Alg.~\ref{alg:visit}), a pre-empting target-closing takeover
(Alg.~\ref{alg:closing}) and a never-idle exploration fallback
(Alg.~\ref{alg:fallback}). Established algorithms (YOLO-World, watershed
segmentation, A$^{\ast}$, Dijkstra, RPT$^{\ast}$, the LLM calls) are invoked by
name and not expanded. Each algorithm is followed by the notation it uses; a
symbol introduced earlier is restated briefly with a pointer to the algorithm
that defines it. Default values are those of the runtime under the Gibson
ObjectNav protocol ($T=500$ actions, forward step $\delta=0.25$\,m, turn
$\Delta\psi=30^{\circ}$, one second per action).

\begin{algorithm}[H]\small
\caption{\textsc{UpdateSceneGraph}: perception and observed-only scene-graph update}
\label{alg:scenegraph}
\Input{frame $(I_t, D_t, \pose_t)$; maps $\Occ$, $\Lmk$; graph $\SG$}
\Output{updated $\Occ$, $\Lmk$, $\SG=(\Rooms,\Doors,\Opens,\Stairs)$}
$\Dets_t \gets \Detect{$I_t$}$ \tcp*[r]{open-vocabulary boxes $(b,\kappa,s)$}
$\Occ \gets \Integrate{$\Occ, D_t, \pose_t$}$ \tcp*[r]{free / occupied / unknown BEV cells}
\ForEach{$(b,\kappa,s)\in\Dets_t$ \KwWith $s\ge s_{\det}$}{
  $(\cen,\rho) \gets \Backproject{$b, D_t, \pose_t$}$ \tcp*[r]{coherent 3-D centroid and footprint radius}
  $\lambda \gets$ nearest $\lambda'\!\in\!\Lmk$ with $\kappa_{\lambda'}{=}\kappa$ \KwAnd
     $\big(\|\cen-\cen_{\lambda'}\|\le d_{\mathrm{dup}}$ \KwOr $\mathrm{IoU}(\cen,\rho;\cen_{\lambda'},\rho_{\lambda'})\ge\iota\big)$
     \tcp*[r]{same class only}
  \lIf{$\lambda$ exists}{$\cen_\lambda \gets$ running mean$(\cen_\lambda,\cen)$; $n_\lambda \gets n_\lambda+1$}
  \lElse{$\Lmk \gets \Lmk\cup\{\lambda{=}(\kappa,\cen,\rho,n{=}1)\}$}
}
$\Lmk^{+} \gets \{\lambda\in\Lmk : n_\lambda\ge n_{\min}\}$ \tcp*[r]{confirmed landmarks; classes never merge}
$\Doors \gets \DoorDet{$\Dets_t,\Occ$}$;\quad $\Stairs\gets$ seen staircases\;
$\Rooms \gets \Watershed{$\mathrm{free}(\Occ)\setminus\mathrm{fp}(\Stairs),\ \Doors\cup\Theta$}$ \tcp*[r]{cut at doors and walked thresholds; sticky ids}
\ForEach{$r\in\Rooms$}{
  $\Omega_r, A_r, F_r \gets$ mask, area, live frontier clusters;\quad $\Lmk_r\gets\{\lambda\in\Lmk^{+}:\cen_\lambda\in\Omega_r\}$\;
  \lIf(\tcp*[f]{weak: 1 class; strong: $\ge2$ classes or a signature object}){a new class entered $\Lmk_r$}{$(\ell_r,\sigma_r)\gets\RoomLLM{$\Lmk_r$}$}
}
\KwRet{$(\Occ,\Lmk,\SG)$}
\end{algorithm}
\begin{notation}{\ref{alg:scenegraph}}
\item[$t$] Index of the current action: one RGB-D frame, one decision and one action per step.
\item[$I_t$, $D_t$] RGB image and depth image captured at action $t$.
\item[$\pose_t$] Agent pose $(\pos_t,\psi_t,\theta_t,f_t)$: planar position $\pos_t\in\mathbb{R}^2$ on the floor, yaw $\psi_t$, camera pitch $\theta_t$ and storey (floor id) $f_t$.
\item[$\Dets_t$] Detections of the frame: boxes $b$ with class $\kappa$ and confidence $s$.
\item[$s_{\det}$] Confidence below which a box is ignored ($0.30$).
\item[$\Occ$] Observed bird's-eye-view occupancy grid of the storey in force; each cell free, occupied or unknown; $\mathrm{free}(\Occ)$ its free cells. One grid per storey.
\item[$(\cen,\rho)$] Back-projected 3-D centroid of a box (from its connected depth support, not a centre-ray depth) and footprint radius (half the box width at its depth).
\item[$\Lmk$] Object-landmark map. A landmark $\lambda=(\kappa_\lambda,\cen_\lambda,\rho_\lambda,n_\lambda)$: class, running-mean centroid, footprint radius, number of frames it was observed in.
\item[$d_{\mathrm{dup}}$, $\iota$] Same-class association rule: centroids within $d_{\mathrm{dup}}$ ($0.70$\,m) \emph{or} footprint discs with intersection-over-union $\ge\iota$ ($0.25$), so the rule scales with object size (a bed re-seen from its other side merges, two chairs $0.3$\,m apart do not). Observations never merge across classes.
\item[$\Lmk^{+}$, $n_{\min}$] Confirmed landmarks: observed in at least $n_{\min}=2$ frames (false-positive guard). Only these feed rooms, labels and nodes.
\item[$\Doors$] Confirmed door detections (position, observation count).
\item[$\Stairs$, $\mathrm{fp}(\Stairs)$] Seen staircases (connectors between storeys) and their footprint cells, which are never room floor.
\item[$\Theta$] Walked thresholds: clearance dips along the agent's trail, used as extra cuts beside the doors.
\item[$\Rooms$, $r$] Rooms of the watershed partition; ids are sticky across re-segmentations.
\item[$\Omega_r$, $A_r$, $F_r$] Room mask (cells), area, and live frontier clusters: accessible unexplored boundary inside the room.
\item[$\Lmk_r$] Confirmed landmarks standing in room $r$: the landmark's cell, else the nearest room floor within its footprint.
\item[$(\ell_r,\sigma_r)$] Room label (\emph{bedroom}, \emph{kitchen}, \dots, or \emph{unknown}) and its strength: \emph{weak} from one object class or a hesitant model, \emph{strong} from $\ge2$ distinct classes or one signature object (bed, toilet, oven). Re-asked whenever a new class appears in the room.
\item[$\SG$] Scene graph $(\Rooms,\Doors,\Opens,\Stairs)$; the openings $\Opens$ are derived at loop points (Alg.~\ref{alg:nodes}).
\item[Routines] \textsc{YOLO-World}: open-vocabulary detector. \textsc{IntegrateDepth}: depth-to-BEV occupancy integration. \textsc{Backproject}: box-to-3-D projection. \textsc{ConfirmedDoors}: door detector with multi-frame confirmation. \textsc{WatershedRooms}: watershed room segmentation cut at doors and thresholds. \textsc{LLM-RoomClassifier}: room-type LLM over the room's objects.
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{RoomFinished}: scan-ledger verdict for room $r$}
\label{alg:finished}
\Input{room $r$: mask $\Omega_r$, area $A_r$, live frontier $F_r$, door flag $\delta_r$; ledger $\Ledger_f$; pose trail $\mathcal{P}_f$}
\Output{verdict $\in\{\varnothing,\ \tsc{inside},\ \tsc{seen},\ \tsc{through},\ \tsc{fragment}\}$}
\lIf(\tcp*[f]{sticky unless the room regrew}){$r$ was finished before \KwAnd $A_r\le\gamma A_r^{\mathrm{then}}$}{\KwRet{earlier verdict}}
\lIf(\tcp*[f]{a completed look-around stood in $r$}){$\exists\,\mathbf{z}\in\Ledger_f:\ \mathbf{z}\in\Omega_r$}{\KwRet{\tsc{inside}}}
\lIf(\tcp*[f]{a scan point saw most of $r$}){$\exists\,\mathbf{z}\in\Ledger_f:\ \Vis{$\mathbf{z},\Omega_r$}\ge\eta$}{\KwRet{\tsc{seen}}}
\If(\tcp*[f]{no accessible unexplored boundary left}){$F_r=\varnothing$}{
  \lIf{$\exists\,\pose\in\mathcal{P}_f:\ \Vis{$\pose,\Omega_r$}\ge\eta$ \KwOr $\big(\pos\in\Omega_r$ \KwAnd $w(\Omega_r)<w_{\mathrm{nar}}\big)$}{\KwRet{\tsc{through}}}
  \lIf(\tcp*[f]{the strip behind a bed, not a bathroom}){$A_r<A_{\mathrm{frag}}$ \KwAnd \KwNot $\delta_r$}{\KwRet{\tsc{fragment}}}
}
\KwRet{$\varnothing$}
\end{algorithm}
\begin{notation}{\ref{alg:finished}}
\item[$r$; $\Omega_r$, $A_r$, $F_r$] Room under judgement, its mask, area and live frontier clusters (Alg.~\ref{alg:scenegraph}).
\item[$\delta_r$] Door flag: whether a confirmed door lies on the room's boundary.
\item[$f$, $\Ledger_f$] Storey in force and its scan ledger: the positions $\mathbf{z}$ at which a full look-around was completed. Remembered by place, not by room id, because the watershed renumbers rooms.
\item[$\mathcal{P}_f$] Trail of poses $\pose$ (positions $\pos$) the agent stood at on storey $f$.
\item[$\Vis{$\cdot,\Omega_r$}$] Fraction of $\Omega_r$ a camera at the given point could see: within depth range, with a clear line of sight through observed free space (unknown cells block sight); for a pose, restricted to its camera cone.
\item[$\eta$] Seen fraction that credits a room to a scan point or a pose ($0.5$).
\item[$w(\Omega_r)$, $w_{\mathrm{nar}}$] Widest clearance inside the room and the narrow bound ($0.9$\,m): a corridor or balcony that the camera cone spans while it is walked.
\item[$A_{\mathrm{frag}}$] Fragment bound ($3$\,m$^2$): a doorless sliver with no frontier is a piece of a neighbour (the strip behind a bed), not a room.
\item[$A_r^{\mathrm{then}}$, $\gamma$] Area at which the earlier verdict was reached and the regrow factor ($1.5$); a verdict is sticky unless the room grows past $\gamma A_r^{\mathrm{then}}$.
\item[verdicts] \tsc{inside}: a scan stood in the room. \tsc{seen}: a scan saw at least $\eta$ of it. \tsc{through}: no frontier left and the room was looked into or walked through. \tsc{fragment}: as above. $\varnothing$: not finished.
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{CandidateNodes}: the node set at a loop point}
\label{alg:nodes}
\Input{$\SG$, $\Occ$, $\Lmk^{+}$, $\Ledger$, $\Rej$; target $\target$; node in force $v$}
\Output{nodes $\Nodes_t$, each with kind $k_i$, entry point $e_i$, status $s_i$; finished rooms $X_t$}
$X_t \gets \{r\in\Rooms\setminus\{v\} : \Finished{$r$}\ne\varnothing\}$ \tcp*[r]{Alg.~\ref{alg:finished}}
drop from $X_t$ any $r$ finished only from outside that holds a home object of $\target$ \tcp*[r]{the toilet behind the jamb}
\lIf(\tcp*[f]{floor exhausted: second look forced}){$\Rooms\setminus X_t=\varnothing$ \KwAnd no exit \KwAnd no reachable frontier}{$X_t\gets\varnothing$}
$\Nodes_t\gets\varnothing$\;
\ForEach{$r\in\Rooms\setminus\{v\}$}{
  $\Nodes_t \gets \Nodes_t\cup\{(r,\ \kind{room},\ e_r{=}\Vantage{$\Omega_r$},\ s_r)\}$ \tcp*[r]{$s_r\in\{$never, entered, scanned(how, ago)$\}$}
}
$\Opens \gets$ exits of the mapped floor: frontier clusters $\varphi\in\Front$ with $|\varphi|\ge n_{\mathrm{op}}$, not an object's shadow, not under the agent, merged within $d_{\mathrm{mrg}}$\;
\ForEach{$o\in\Opens$ \KwNot yet peeked or abandoned}{
  $\Nodes_t \gets \Nodes_t\cup\{(o,\ \kind{opening},\ e_o{=}\text{threshold},\ \text{room of origin},\ \text{door seen?},\ \text{classes glimpsed through } o)\}$\;
}
\ForEach{$\lambda\in\Lmk^{+}$ \KwWith $\kappa_\lambda$ accepted by $\target$, \KwNot inspected, $\cen_\lambda\notin\Rej$}{
  $\Nodes_t \gets \Nodes_t\cup\{(\lambda,\ \kind{landmark},\ e_\lambda{=}\text{standoff at } d_{\mathrm{L}} \text{ on the agent's side})\}$ \tcp*[r]{fixed $p_\lambda{=}p_{\mathrm{L}}$, no oracle call}
}
\lIf(\tcp*[f]{multi-storey only}){storey changes allowed}{add every $k\in\Stairs$ as $(k,\ \kind{stairs},\ e_k{=}\text{foot of the flight})$}
\KwRet{$(\Nodes_t, X_t)$}
\end{algorithm}
\begin{notation}{\ref{alg:nodes}}
\item[$\SG$, $\Occ$, $\Lmk^{+}$] Scene graph, occupancy grid and confirmed landmarks (Alg.~\ref{alg:scenegraph}).
\item[$\Ledger$] Scan ledgers of all storeys (Alg.~\ref{alg:finished}).
\item[$\Rej$] Rejection memory: anchors of released target candidates, each with a radius (Alg.~\ref{alg:closing}); a landmark inside one is not offered.
\item[$\target$] Target category; it \emph{accepts} a detector class $\kappa$ when $\kappa$ is one of its aliases.
\item[$v$] Node in force; never excluded in the middle of its visit.
\item[$\Nodes_t$] Candidate nodes. Node $i$ has kind $k_i\in\{\kind{room},\kind{opening},\kind{landmark},\kind{stairs}\}$, entry point $e_i$ (where the agent is sent) and status $s_i$.
\item[$X_t$] Finished rooms, withheld from the order while the pass verdict is \tsc{first} (Alg.~\ref{alg:main}); emptied when the floor is exhausted so that a second look becomes possible.
\item[home object] Object that marks the room the target lives in (a sink or shower for a toilet); such a room stays a node whatever its label or verdict.
\item[$s_r$] Room status shown to the oracle: never entered, entered, or scanned (how, and how long ago).
\item[$e_r=\Vantage{$\Omega_r$}$] Entry point of a room: the reachable interior cell of greatest clearance, $\arg\max_{c\in\Omega_r}\mathrm{DT}_{\Omega_r}(c)$, with $\mathrm{DT}$ the Euclidean distance transform of the mask; on a revisit, outside discs of radius $d_{\mathrm{rv}}$ ($1.5$\,m) around earlier scan points.
\item[$\Front$, $\varphi$] Accessible-frontier inventory: clusters $\varphi$ of unexplored boundary reachable at the body clearance.
\item[$\Opens$, $o$] Openings: genuine exits of the mapped floor -- clusters of at least $n_{\mathrm{op}}=8$ cells that are not an \emph{object's shadow} (the strip of unknown behind furniture, no longer than the object could cast) and not the blind disc under the agent -- merged within $d_{\mathrm{mrg}}=1.5$\,m. Retired for the storey once peeked or found unreachable.
\item[$e_o$] The opening's threshold cell on the agent's side.
\item[glimpsed classes] Confirmed landmarks seen through the opening: in its cone ($\pm35^{\circ}$), beyond its threshold, within $3$\,m.
\item[$e_\lambda$, $d_{\mathrm{L}}$, $p_{\mathrm{L}}$] Target-landmark node: a confirmed landmark whose class the target accepts, entered at a standoff of $d_{\mathrm{L}}=1.5$\,m on the agent's side and valued at the fixed $p_{\mathrm{L}}=0.85$ without an oracle call.
\item[$k\in\Stairs$, $e_k$] Seen staircase and its entry, the foot of the flight on this storey's map (only when storey changes are allowed).
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{NodeOracle}: $p_i = P(\text{target found}\mid\text{go to node } i \text{ next})$}
\label{alg:oracle}
\Input{target $\target$; nodes $\Nodes_t$; $\SG$, $\Ledger$; action $t$, budget $T$; cached prompt/reply $(\mathcal{U}^{-},\mathcal{Y}^{-})$}
\Output{$\mathbf{p}=(p_i)_{i\in\Nodes_t}$; pass verdict $\varpi\in\{\tsc{first},\tsc{second}\}$}
$H \gets \HouseLine{$\SG,\Ledger$}$: room types found with status, \#unidentified rooms, \#openings not looked into, frontier left?, actions used $\approx\lfloor t/\Delta_B\rfloor\Delta_B$ of $T$\;
\ForEach{$i\in\Nodes_t$ \KwWith $k_i\ne\kind{landmark}$}{
  $u_i\gets$ one line: kind, $\ell_i$ (tentative if $\sigma_i$ weak), $A_i$, $|F_i|$, $s_i$, $\tau_i$, $\Lmk_i$;\ opening: origin room, door?, glimpsed classes;\ stairs: direction, other storey's record\;
}
$\mathcal{U}\gets(\target, H, \{u_i\})$\;
\lIf(\tcp*[f]{byte-identical prompt $\Rightarrow$ no call}){$\mathcal{U}=\mathcal{U}^{-}$}{$\mathcal{Y}\gets\mathcal{Y}^{-}$}
\lElse(\tcp*[f]{reasons on \emph{home/house/stage/pass} first; no distance}){$\mathcal{Y}\gets\LLM{rules, $\mathcal{U}$}$}
\lForEach(\tcp*[f]{independent estimates, \emph{not} renormalised}){$i$ scored $y_i$ in $\mathcal{Y}$}{$p_i \gets \min\!\big(\max(y_i/100,\,0),\ p_{\max}\big)$}
\lForEach(\tcp*[f]{forgotten, not ruled out}){$i$ omitted from $\mathcal{Y}$}{$p_i\gets p_{\mathrm{omit}}$}
discard invented ids;\ \lIf{no usable node remains}{repair the schema once, else back off (Alg.~\ref{alg:fallback})}
$p_{\varnothing}\gets\prod_{i}(1-p_i)$ \tcp*[r]{mass in none of them, under RPT$^{\ast}$'s independence approximation}
$\varpi\gets$ the \emph{pass} verdict of $\mathcal{Y}$ (\tsc{first} if absent)\;
\KwRet{$(\mathbf{p},\varpi)$}
\end{algorithm}
\begin{notation}{\ref{alg:oracle}}
\item[$\target$, $\Nodes_t$] Target category and candidate nodes (Alg.~\ref{alg:nodes}); $k_i$ is the kind of node $i$.
\item[$\SG$, $\Ledger$] Scene graph (Alg.~\ref{alg:scenegraph}) and scan ledger (Alg.~\ref{alg:finished}), read for the house line.
\item[$t$, $T$] Current action and the episode's action budget ($500$).
\item[$H$] \emph{House line}: the room types found so far with their status, the number of unidentified rooms, the openings not yet looked into, whether any frontier is left, and the budget spent.
\item[$\Delta_B$] Budget step ($25$ actions): effort and budget are shown in coarse steps so that an unchanged map yields an unchanged prompt.
\item[$u_i$] Prompt line of node $i$: kind, label $\ell_i$ (marked tentative if its strength $\sigma_i$ is weak), area $A_i$, frontier count $|F_i|$, status $s_i$, time searched $\tau_i$, objects $\Lmk_i$; for an opening its origin room, door and glimpsed classes; for stairs the direction and the other storey's record. Target landmarks are not shown.
\item[$\mathcal{U}$, $\mathcal{Y}$] Prompt and structured (JSON) reply; $(\mathcal{U}^{-},\mathcal{Y}^{-})$ is the cached pair, reused when the prompt is byte-identical.
\item[rules] System prompt: where the target normally lives, what the rooms found imply about the unidentified ones, how the budget weighs exploration against a second look -- and \emph{not} distance, which Alg.~\ref{alg:order} charges.
\item[$y_i$] The model's score for node $i$, in percent.
\item[$p_i$, $\mathbf{p}$] Probability that going to node $i$ \emph{next} finds the target; the estimates are independent per node and never normalised to sum to one. $p_{\max}=1-10^{-6}$ is the ceiling; $p_{\mathrm{omit}}=0.01$ the share given to a node the model forgot.
\item[$p_{\varnothing}$] Probability that the target is in none of the nodes, $\prod_i(1-p_i)$ (independent-terminal approximation, as in RPT$^{\ast}$).
\item[$\varpi$] Pass verdict: \tsc{first} while an unexplored place is still worth looking at before any scanned room is re-checked; \tsc{second} once the home is covered and a detector miss becomes the likelier explanation.
\item[Routines] \textsc{HouseLine}: renders $H$ from the graph and the ledger. \textsc{LLM-Reason}: one call to the reasoning model, answered in a fixed JSON schema.
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{OrderNodes}: cost-aware visit order (HPP-PT instance for RPT$^{\ast}$)}
\label{alg:order}
\Input{nodes $\Nodes_t$ with entries $e_i$ and probabilities $p_i$; $\Occ$; pose $\pos_t$; cruise speed $\nu$; service times $\tau^{\mathrm{svc}}_i$; stair leaves $\ell_i$}
\Output{visit order $\pi=(\pi_1,\dots,\pi_m)$}
\lForEach(\tcp*[f]{never a fabricated arc weight}){$i\in\Nodes_t$}{$e_i\gets\Snap{$e_i, r_{\mathrm{snap}}$}$; drop $i$ if none}
$V\gets\{e_i\}\cup\{e_0{=}\pos_t\}$ \tcp*[r]{the depot is the agent's own cell}
$d\gets\Dijkstra{unit-weight 8-graph of passable cells, $V$}$ \tcp*[r]{all pairs in one sweep; metric, in metres}
keep the nodes $d$ connects to $e_0$; cap to $m\le m_{\max}$ (nearest first, then likeliest)\;
$C_{uv}\gets \big(d(e_u,e_v)+\ell_u+\ell_v\big)/\nu+\tau^{\mathrm{svc}}_v$ \tcp*[r]{service on entering arcs; stair leaf on every incident arc}
$p_i\gets\min(p_i,\,1-\epsilon)$ \tcp*[r]{RPT$^{\ast}$ divides by $1-p$}
$\pi\gets\RPT{$V, C, \mathbf{p}, e_0$}$ \tcp*[r]{$\textstyle\min_\pi J(\pi)=\sum_{i=0}^{m-1}\Big[\prod_{j=1}^{i}(1-p_{\pi_j})\Big]C_{\pi_i\pi_{i+1}}$, $\pi_0=e_0$}
\KwRet{$\pi$} \tcp*[r]{the whole order; committed until the head's visit ends}
\end{algorithm}
\begin{notation}{\ref{alg:order}}
\item[$\Nodes_t$, $e_i$, $p_i$] Nodes with their entry points (Alg.~\ref{alg:nodes}) and probabilities (Alg.~\ref{alg:oracle}; $p_{\mathrm{L}}$ for landmark nodes).
\item[$\Occ$, $\pos_t$] Occupancy grid (Alg.~\ref{alg:scenegraph}) and the agent's planar position.
\item[$r_{\mathrm{snap}}$] Radius ($2.0$\,m) within which a blocked entry point is moved onto a passable cell; a node with none is dropped, never priced with a made-up weight.
\item[$V$, $e_0$] Vertex set of the routing instance and the depot: the agent's own cell, with probability $0$.
\item[$d(e_u,e_v)$] Geodesic distance in metres on the unit-weight 8-connected graph of passable cells; binary weights keep the matrix metric and symmetric, which RPT$^{\ast}$'s dominance pruning requires.
\item[$m$, $m_{\max}$] Number of nodes in the solve and its cap ($12$): the nearest node is always kept, the rest by probability.
\item[$\nu$] Cruise speed: forward step per action time ($0.25$\,m\,s$^{-1}$).
\item[$\tau^{\mathrm{svc}}_i$] Service time of node $i$, folded into every arc entering it: $\tau_{\mathrm{scan}}=22$\,s for a room, $\tau_{\mathrm{peek}}=4$\,s for an opening or landmark. Folding preserves the triangle inequality.
\item[$\ell_i$] Stair leaf in metres: flight length plus the fixed cost of a storey change, charged on every arc into \emph{and} out of a stair node ($0$ for other nodes).
\item[$C_{uv}$] Arc cost in seconds from node $u$ to node $v$.
\item[$\epsilon$] Probability clamp ($10^{-6}$): RPT$^{\ast}$'s heuristic divides by $1-p$.
\item[$\pi$, $J(\pi)$] Visit order $(\pi_1,\dots,\pi_m)$ with $\pi_0=e_0$, and its expected time-to-find: each leg is paid only if the target was not found at the nodes before it.
\item[Routines] \textsc{SnapToPassable}: nearest passable cell within a radius. \textsc{MultiSourceDijkstra}: all-pairs shortest paths in one sweep. \textsc{RPT}$^{\ast}$: exact solver of the Hamiltonian path problem with probabilistic terminals.
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{Visit}: the node in force, one action per call}
\label{alg:visit}
\Input{node $v$ with kind $k_v$ and entry $e_v$; $\Occ$; pose $\pose_t$; ledger $\Ledger$; visit counter $n_v$}
\Output{action $a_t$, or a release verdict $\in\{\tsc{exhausted},\tsc{unreachable},\tsc{budget},\tsc{reclassified},\tsc{traversed}\}$}
\lIf{$n_v>B_{k_v}$}{\KwRet{\tsc{released}: budget}}
\lIf(\tcp*[f]{re-plan on the new fact}){$k_v=\kind{room}$ \KwAnd a new class renamed $v$}{\KwRet{\tsc{released}: reclassified}}
\If(\tcp*[f]{transit}){$\|\pos_t-e_v\|>d_{\mathrm{arr}}$}{
  \lIf(\tcp*[f]{$\arg\max_{c\in\Omega_v}\mathrm{DT}_{\Omega_v}(c)$, re-read as the map grows}){$k_v=\kind{room}$}{$e_v\gets\Vantage{$\Omega_v$}$}
  $a_t\gets$ next step of $\AStar{$\Occ,\pos_t,e_v$}$ on the room-confined map\;
  \lIf{no route}{\KwRet{\tsc{released}: unreachable}} \lElse{\KwRet{$a_t$}}
}
\Switch{$k_v$}{
  \uCase(\tcp*[f]{the look-around: the camera's ray ends at the nearest object}){\kind{room}}{
    \lIf{yaw swept $<2\pi$ since arrival}{\KwRet{turn by $\Delta\psi$}}
    $\Ledger_f\gets\Ledger_f\cup\{\pos_t\}$;\ \KwRet{\tsc{released}: exhausted} \tcp*[r]{finished for the episode}
  }
  \uCase(\tcp*[f]{peek into the unknown / at the landmark}){\kind{opening}, \kind{landmark}}{
    \lIf{headings $h_v-\Delta\psi,\ h_v,\ h_v+\Delta\psi$ not all looked at}{\KwRet{turn toward the next}}
    retire $v$ for the storey;\ \KwRet{\tsc{released}: exhausted} \tcp*[r]{its floor joins the partition}
  }
  \Case{\kind{stairs}}{
    hand control to \Traverse{$v$}; on a storey first set foot on: \Rotate{} and $\Ledger_{f'}\gets\Ledger_{f'}\cup\{\pos\}$;\ \KwRet{\tsc{released}: traversed}\;
  }
}
\end{algorithm}
\begin{notation}{\ref{alg:visit}}
\item[$v$, $k_v$, $e_v$] Node in force, its kind and its entry point (Alg.~\ref{alg:nodes}); $\Omega_v$ its mask when it is a room; $h_v$ the heading of an opening (toward the unknown) or toward a landmark.
\item[$\Occ$, $\pose_t$, $\pos_t$] Occupancy grid, pose and planar position (Alg.~\ref{alg:scenegraph}).
\item[$\Ledger_f$, $\Ledger_{f'}$] Scan ledger of the storey in force and of the storey $f'$ arrived at by stairs (Alg.~\ref{alg:finished}).
\item[$n_v$, $B_{k_v}$] Actions spent on this visit and the bound per kind: $B_{\mathrm{room}}=36$ for a scan, $B_{\mathrm{peek}}=30$ for a peek.
\item[$d_{\mathrm{arr}}$] Arrival radius at the entry point ($0.6$\,m).
\item[$\mathrm{DT}_{\Omega_v}(c)$] Euclidean distance transform of the room mask at cell $c$; its maximiser is the vantage point, re-read every action as the map grows.
\item[room-confined map] Planning copy of $\Occ$ with every other room blocked, so the approach stays inside the room in force.
\item[$\Delta\psi$] Yaw of one turn action ($30^{\circ}$); a scan is a full $2\pi$ sweep measured on the pose.
\item[$a_t$] The action emitted by this call.
\item[release verdicts] \tsc{exhausted}: scan or peek complete (productive, not repeatable). \tsc{unreachable}: no route. \tsc{budget}: $n_v>B_{k_v}$. \tsc{reclassified}: a new object class renamed the room. \tsc{traversed}: the climb has begun. Every verdict returns control to a loop point (Alg.~\ref{alg:main}).
\item[Routines] \textsc{Vantage}: see $\mathrm{DT}$. \textsc{A}$^{\ast}$: weighted, clearance-qualified grid planner. \textsc{TraverseStairs}: building coordinator that approaches the stair foot and climbs. \textsc{FullRotation}: one $2\pi$ turn in place (the warm-up).
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{TargetClosing}: verify, approach, inspect; \tsc{stop} or release}
\label{alg:closing}
\Input{detections $\Dets_t$ with 3-D projections; pose $\pose_t$; $\Occ$; rejection memory $\Rej$; phase $\phi$; memory $\mem$; streak $n_c$}
\Output{action $a_t$; \tsc{stop}; or \tsc{released} (control returns to the search)}
$\mathcal{C}\gets\{(b,\kappa,s,\cen)\in\Dets_t : \target \text{ accepts } \kappa,\ s\ge s_{\mathrm{lock}},\ b \text{ not clipped at the image border},\ \cen\notin\Rej\}$\;
\If(\tcp*[f]{takeover: exploration is suspended}){$\phi=\varnothing$ \KwAnd $\mathcal{C}\ne\varnothing$}{
  $\cen_0\gets$ most confident candidate;\ $\mem\gets\cen_0$;\ $n_c\gets 0$;\ $\phi\gets\tsc{verify}$\;
  \lIf(\tcp*[f]{re-verify from a wider view}){$\|\pos_t-\cen_0\|<d_{\mathrm{sud}}$}{\BackOff{}: retreat $n_b$ steps, face $\cen_0$}
}
\uIf{some $\cen\in\mathcal{C}$ has $\|\cen-\mem\|\le r_{\mathrm{as}}$}{
  $\mem\gets$ precision-weighted update, $w=1/\sigma(\|\pos_t-\cen\|)^2$;\ $n_c\gets n_c+1$ \tcp*[r]{3-D target memory}
}
\lElseIf{$\phi=\tsc{verify}$}{$n_c\gets 0$}
\Switch{$\phi$}{
  \uCase{\tsc{verify}}{
    \lIf(\tcp*[f]{locked}){$n_c\ge k_c$ (or $k'_c$, from viewpoints $\ge\beta$ apart, if context-suspect)}{$\phi\gets\tsc{close}$}
    \lElseIf{$>n_{\mathrm{ver}}$ actions in \tsc{verify}}{\KwRet{\Release{$\mem,\varrho$}}}
    \lElse{\KwRet{face $\mem$ (sweep if out of view)}}
  }
  \uCase{\tsc{close}}{
    \lIf(\tcp*[f]{the map outvotes the lock}){$\Lmk^{+}$ now holds another class at $\cen_0$}{\KwRet{\Release{$\mem,2\varrho$}}}
    \lIf{$\|\pos_t-\mem\|\le d_{\mathrm{term}}$}{$\phi\gets\tsc{inspect}$}
    \lElse(\tcp*[f]{no path or stalled: release}){\KwRet{step of $\AStar{$\Occ,\pos_t,\mathrm{standoff}(\mem)$}$; pitch to $\mem$ within $d_{\mathrm{look}}$}}
  }
  \Case{\tsc{inspect}}{
    $\hat d\gets$ freshly measured near-edge range to the target, if seen this frame\;
    \lIf{seen fresh \KwAnd $\hat d\le d_{\mathrm{term}}+\epsilon_d$}{\KwRet{\tsc{stop}}}
    \lIf{$n_{\mathrm{insp}}$ views spent \KwAnd it was seen in range from here}{\KwRet{\tsc{stop}}}
    \lIf(\tcp*[f]{wrong lock}){$n_{\mathrm{insp}}$ views spent \KwAnd never seen from here}{\KwRet{\Release{$\mem,2\varrho$}}}
    \KwRet{next bounded yaw/pitch view centred on $\mem$}\;
  }
}
\Fn{\Release{$\cen,\varrho'$}}{$\Rej\gets\Rej\cup\{(\cen,f,\varrho')\}$;\ $\phi\gets\varnothing$;\ no new takeover for $n_{\mathrm{cool}}$ actions;\ \KwRet{\tsc{released}}}
\end{algorithm}
\begin{notation}{\ref{alg:closing}}
\item[$\Dets_t$] This frame's detections with their 3-D projections $(b,\kappa,s,\cen)$: box, class, confidence, centroid (Alg.~\ref{alg:scenegraph}).
\item[$\pose_t$, $\pos_t$, $\Occ$] Pose, planar position and occupancy grid (Alg.~\ref{alg:scenegraph}).
\item[$\Rej$] Rejection memory: anchors $(\cen,f,\varrho')$ with storey $f$ and radius $\varrho'$; no takeover starts on a candidate inside one.
\item[$\phi$] Closing phase: $\varnothing$ (inactive), \tsc{verify}, \tsc{close}, \tsc{inspect}; terminal outcomes \tsc{stop} and \tsc{released}.
\item[$\mathcal{C}$, $s_{\mathrm{lock}}$] Candidate boxes of the target this frame: confidence at least $s_{\mathrm{lock}}=0.50$, a coherent depth projection, not a sliver at the image edge, not inside $\Rej$.
\item[$\target$] Target category; \emph{accepts} $\kappa$ as in Alg.~\ref{alg:nodes}.
\item[$\cen_0$] Anchor: the candidate's first centroid, kept for association and for $\Rej$.
\item[$\mem$] Fused 3-D target memory: precision-weighted running mean of the associated projections, each with weight $w=1/\sigma(d)^2$, $\sigma(d)=\sigma_0+\sigma_1 d$ at range $d$ ($\sigma_0=0.03$\,m, $\sigma_1=0.03$).
\item[$r_{\mathrm{as}}$] Association radius between a new projection and $\mem$ ($0.50$\,m, growing slowly with range).
\item[$n_c$, $k_c$, $k'_c$, $\beta$] Streak of consecutive frames seeing the candidate and the frames needed to lock ($k_c=2$); $k'_c=4$ frames from viewpoints at least $\beta=0.30$\,m apart when the context check doubts the box (e.g.\ a bed-sized box in a kitchen).
\item[$d_{\mathrm{sud}}$, $n_b$] Sudden-proximity range ($1.30$\,m) below which the back-off manoeuvre first retreats $n_b=3$ steps over known-free cells.
\item[$n_{\mathrm{ver}}$] Verification bound ($12$ actions); an unverified candidate is released.
\item[$\Lmk^{+}$] Confirmed landmarks (Alg.~\ref{alg:scenegraph}); a lock whose anchor the map has since confirmed as another class is released.
\item[$\mathrm{standoff}(\mem)$] Approach goal: a reachable cell beside the target at terminal range.
\item[$d_{\mathrm{term}}$, $\epsilon_d$] Terminal range for \tsc{stop} ($1.00$\,m) and its tolerance ($0.05$\,m).
\item[$d_{\mathrm{look}}$] Range ($1.30$\,m) within which the camera pitch follows the target's height (down at a toilet, up at a wall-mounted television).
\item[$\hat d$] Freshly measured near-edge range of the detected surface, which is what the success radius is measured to; never the remembered range alone.
\item[$n_{\mathrm{insp}}$] Inspection bound ($24$ actions) of stationary yaw/pitch views; the whole closing is bounded to $160$ actions.
\item[$\varrho$, $n_{\mathrm{cool}}$] Rejection radius ($1.0$\,m, doubled when a lock failed inspection or the map outvoted it) and the cooldown before another takeover may start ($15$ actions).
\item[$a_t$] The action emitted by this call.
\item[Routines] \textsc{BackOff}: the backing manoeuvre. \textsc{A}$^{\ast}$: planner toward the standoff. \textsc{Release}: records the anchor in $\Rej$ and hands the action back to the search.
\end{notation}

\begin{algorithm}[H]\small
\caption{\textsc{ExplorationFallback}: the search never idles}
\label{alg:fallback}
\Input{$\Occ$, $\SG$; ranked accessible-frontier inventory $\Front$; retired goals; floor state}
\Output{an action that moves (a hold only as the last rung)}
\lIf{a model service failed}{arm a doubling back-off; do not ask it until the back-off ends}
$\Front_{\mathrm{exit}}\gets\{\varphi\in\Front:\ \text{reachable},\ \KwNot \text{ an object's shadow},\ \KwNot \text{ in a type-excluded room}\}$\;
\lIf(\tcp*[f]{rung 1: the best exit on the floor}){$\Front_{\mathrm{exit}}\ne\varnothing$}{\KwRet{A$^{\ast}$ step toward $\arg\max_{\varphi\in\Front_{\mathrm{exit}}} U(\varphi)$}}
\lIf(\tcp*[f]{rung 2}){floor exhausted \KwAnd storey changes allowed}{\KwRet{\Traverse{explicit up/down rule}}}
\lIf(\tcp*[f]{rung 3: still unknown space}){$\Front\setminus\Front_{\mathrm{exit}}\ne\varnothing$}{\KwRet{step toward the nearest}}
\lIf(\tcp*[f]{rung 4}){retired goals remain}{\KwRet{step toward the nearest}}
\lIf(\tcp*[f]{rung 5: a new vantage point}){a reachable known cell exists}{\KwRet{relocate to the farthest one}}
\lIf(\tcp*[f]{rung 6: blind disc underfoot}){footing not yet swept here}{\KwRet{\tsc{look\_down}, one rotation, \tsc{look\_up}}}
\KwRet{hold} \tcp*[r]{rung 7: every rung spent at this spot}
\end{algorithm}
\begin{notation}{\ref{alg:fallback}}
\item[$\Occ$, $\SG$] Occupancy grid and scene graph (Alg.~\ref{alg:scenegraph}).
\item[$\Front$, $\varphi$] Accessible-frontier inventory and one of its clusters (Alg.~\ref{alg:nodes}), ranked by utility $U(\varphi)$: cluster size, discounted by geodesic travel and heading change.
\item[$\Front_{\mathrm{exit}}$] Frontiers that lead somewhere: reachable, not an object's shadow (Alg.~\ref{alg:nodes}), not in a room whose identified type excludes the target.
\item[retired goals] Frontier goals given up after a transient plan failure; still unknown space.
\item[back-off] A failed LLM or detector service is not asked again for a doubling number of actions; this algorithm carries the search meanwhile.
\item[footing sweep] \tsc{look\_down}, one rotation, \tsc{look\_up}: maps the blind disc under the camera that makes a stationary agent's surroundings read as impassable; done once per spot.
\item[hold] An in-place turn; the only action that changes anything once every rung is spent.
\item[Routines] \textsc{A}$^{\ast}$ and \textsc{TraverseStairs} as in Alg.~\ref{alg:visit}.
\end{notation}

\begin{algorithm}[H]\small
\caption{LLM-guided room search with target closing (main loop)}
\label{alg:main}
\Input{target category $\target$; RGB-D and pose stream $(I_t,D_t,\pose_t)$; action budget $T$}
\Output{\tsc{stop} at the target (\tsc{found}), or \tsc{not found} when the budget is spent}
\State{$\Occ,\Lmk,\SG\gets\varnothing$;\ $\Ledger,\Rej\gets\varnothing$;\ node in force $v\gets\varnothing$;\ pass $\varpi\gets\tsc{first}$}
\Rotate{} in place;\ $\Ledger_{f_0}\gets\Ledger_{f_0}\cup\{\pos_0\}$ \tcp*[r]{warm-up: the spawn room is scanned first}
\For{$t\gets 0$ \KwTo $T-1$}{
  $(\Occ,\Lmk,\SG)\gets\UpdateSG{$I_t,D_t,\pose_t$}$ \tcp*[r]{Alg.~\ref{alg:scenegraph}, every action}
  \If(\tcp*[f]{Alg.~\ref{alg:closing} pre-empts everything}){closing active \KwOr a candidate of $\target$ with a 3-D projection appears}{
    $a_t\gets\Closing{$\Dets_t,\pose_t,\Occ,\Rej$}$\;
    \lIf{$a_t=\tsc{stop}$}{\KwRet{\tsc{found}}}
    \lIf(\tcp*[f]{resume at a loop point, with $\Rej$ grown}){$a_t=\tsc{released}$}{$v\gets\varnothing$}
    \lElse{\Execute{$a_t$};\ \KwContinue}
  }
  $a_t\gets\varnothing$\;
  \Repeat(\tcp*[f]{a released node is replaced on this same action}){$a_t\ne\varnothing$ \KwOr $v=\varnothing$, at most $n_{\mathrm{rnd}}$ rounds}{
    \If(\tcp*[f]{loop point}){$v=\varnothing$}{
      $(\Nodes_t,X_t)\gets\Candidates{$\SG,\Occ,\Lmk^{+},\Ledger,\Rej,\target,v$}$ \tcp*[r]{Alg.~\ref{alg:nodes}}
      \If{$\Nodes_t$ holds an unfinished room, an exit or a staircase}{
        $(\mathbf{p},\varpi)\gets\Oracle{$\target,\Nodes_t,\SG,\Ledger,t,T$}$ \tcp*[r]{Alg.~\ref{alg:oracle}: one reasoning call per loop point}
        \lIf(\tcp*[f]{first pass: no second looks yet}){$\varpi=\tsc{first}$}{$\Nodes_t\gets\Nodes_t\setminus X_t$}
        $p_\lambda\gets p_{\mathrm{L}}$ for landmark nodes;\ $\pi\gets\Order{$\Nodes_t,\mathbf{p},\pos_t$}$;\ $v\gets\pi_1$ \tcp*[r]{Alg.~\ref{alg:order}}
      }
    }
    \If{$v\ne\varnothing$}{
      $a_t\gets\Visit{$v$}$ \tcp*[r]{Alg.~\ref{alg:visit}: transit by A$^{\ast}$, then scan / peek / climb}
      \lIf{$a_t$ is a release verdict}{$v\gets\varnothing$;\ $a_t\gets\varnothing$}
    }
  }
  \lIf(\tcp*[f]{Alg.~\ref{alg:fallback}: no node, no route, or LLM back-off}){$a_t=\varnothing$}{$a_t\gets\Fallback{$\Occ,\SG$}$}
  \Execute{$a_t$}\;
}
\KwRet{\tsc{not found}}
\end{algorithm}
\begin{notation}{\ref{alg:main}}
\item[$\target$] Target object category (the query, e.g.\ \emph{toilet}).
\item[$(I_t,D_t,\pose_t)$] RGB image, depth image and pose at action $t$ (Alg.~\ref{alg:scenegraph}).
\item[$t$, $T$] Action index and the action budget ($500$).
\item[$\Occ$, $\Lmk$, $\SG$] Occupancy grid, landmark map and scene graph (Alg.~\ref{alg:scenegraph}); $\Lmk^{+}$ the confirmed landmarks.
\item[$\Ledger$, $\Rej$] Scan ledger (Alg.~\ref{alg:finished}) and rejection memory (Alg.~\ref{alg:closing}); $\Ledger_{f_0}$ is the ledger of the spawn storey $f_0$ and $\pos_0$ the spawn position.
\item[$v$] Node in force; $v=\varnothing$ is a \emph{loop point}: nothing is committed and the plan is remade now.
\item[$\varpi$] Pass verdict of the last oracle call (Alg.~\ref{alg:oracle}).
\item[$\Dets_t$] This frame's detections, read by the closing (Alg.~\ref{alg:closing}).
\item[$a_t$] Action: \textsc{move\_forward} ($\delta=0.25$\,m), \textsc{turn\_left/right} ($\Delta\psi=30^{\circ}$), \textsc{look\_up/down}, \textsc{stop}; $\varnothing$ while undecided.
\item[$\Nodes_t$, $X_t$] Candidate nodes and finished rooms (Alg.~\ref{alg:nodes}).
\item[$\mathbf{p}$, $p_\lambda$, $p_{\mathrm{L}}$] Oracle probabilities (Alg.~\ref{alg:oracle}); landmark nodes take the fixed $p_{\mathrm{L}}=0.85$.
\item[$\pi$, $\pi_1$] Visit order and its head (Alg.~\ref{alg:order}).
\item[$n_{\mathrm{rnd}}$] Bound ($5$) on same-action re-plans (release $\to$ re-order $\to$ transit) before the fallback takes the action.
\item[outcomes] \tsc{found}: the closing issued \tsc{stop}. \tsc{not found}: the budget ran out. \tsc{released}: the closing gave its candidate back and the search resumes.
\item[Routines] \textsc{FullRotation}: the warm-up, $12$ turns of $30^{\circ}$. \textsc{Execute}: send the action to the robot or simulator and advance $t$.
\end{notation}
